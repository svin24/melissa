# Installed into the container image on top of fedora (no weak deps). One per line; '#' starts a comment.
systemd             # PID 1; systemd-socket-proxyd for the UI forward; sysusers/tmpfiles
# No systemd-udev: --privileged shares the host's /dev and /sys, so a udevd inside would replay "add"
# for every host device at each start (desktops saw USB drives as just plugged in) and its
# systemd-sysctl would write the host's kernel settings. Nothing inside needs it: the host's udev
# makes the device nodes and /dev/disk/by-* links. Its units are masked too (Containerfile, lint).
bubblewrap          # sandbox for agent CLIs (crates/agent/src/sandbox.rs)
nftables            # UI forward filter (ui-forward-filter); agent egress
iptables-nft        # k3s (kube-proxy, flannel, network policy) uses it from PATH; its bundled one
                    # falls back to legacy iptables, which needs the ip_tables module (absent)
iproute             # k3s-prepare (node IP), ui-forward-filter (default route)
util-linux          # mount, nsenter (k3s, smoke test); unshare, nsenter, setpriv (agent users)
shadow-utils        # newuidmap/newgidmap: each agent sandbox its own host user (sandbox/users.rs)
polkit              # lets the server restart itself (Settings → Restart; 50-melissa.rules)
ca-certificates     # HTTPS for agent installers and the speech model download (Settings → Voice)
libstdc++           # whisper.cpp's CLI (voice input, /usr/lib/melissa/whisper)
curl                # agent installers (melissa-setup)
sqlite              # backups: a consistent copy of k3s's datastore (VACUUM INTO; crates/backup/src/k3s.rs)
nodejs24            # Node.js runtime for npm-only agent CLIs (Gemini CLI, Pi); ~73 MiB with npm
nodejs24-npm        # melissa-setup installs those packages (pinned version + integrity) into the agents' home
tar
gzip
unzip
xz
findutils
procps-ng
which
kmod                # modprobe in k3s.service (harmless no-op without /lib/modules)
